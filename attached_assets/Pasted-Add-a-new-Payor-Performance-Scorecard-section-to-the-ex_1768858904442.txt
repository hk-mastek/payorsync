1. Data Source Mapping
Use existing data to calculate metrics. Do NOT create duplicate payor or claims data.
EXISTING DATA → METRIC CALCULATION:

Variance Records (2,250 existing records):
├── Status = "Open" or "In Progress" → Used for A/R calculations
├── Status = "Resolved" → Used for collection rate calculations
├── Status = "Written Off" → Used for write-off rate calculations
├── Variance Amount → Used for underpayment detection
├── Aging Bucket → Used for Days in A/R
├── Root Cause = "Authorization Issues" or denial-related → Denial rate proxy
└── Payor Name/Type → Group by payor

Write-Off Records (850 existing records):
├── Write-Off Amount → Contract compliance issues
├── Write-Off Cause → Denial vs collection issues
├── Days to Write-Off → A/R aging indicator
└── Payor Name → Group by payor

Payor Records (75 existing records):
├── Payor Name → Primary key
├── Payor Type → Grouping
└── State/Region → Geographic analysis

DERIVED METRICS FROM EXISTING DATA:

1. Denial Rate = 
   (Variances with Root Cause in ['Authorization Issues', 'Medical Necessity', 'Coding Errors'] 
   ÷ Total Variances per Payor) × 100

2. Net Collection Rate = 
   ((Total Billed - Total Variance Amount - Write-Offs) ÷ Total Billed) × 100
   
3. Days in A/R = 
   Weighted average of Aging Bucket midpoints for open variances
   
4. Clean Claim Rate = 
   (Variances Resolved without rework ÷ Total Variances) × 100
   OR 100% - Denial Rate - Rework Rate
   
5. Contract Compliance Rate = 
   ((Total Expected - Underpayment Variances) ÷ Total Expected) × 100
   Use variances with Root Cause = 'Contractual Disputes'

javascript// Group existing data by payor
const calculatePayorMetrics = (variances, writeOffs, payors) => {
  return payors.map(payor => {
    const payorVariances = variances.filter(v => v.payorName === payor.payorName);
    const payorWriteOffs = writeOffs.filter(w => w.payorName === payor.payorName);
    
    // Denial Rate: Authorization + Medical Necessity + Coding variances
    const denialVariances = payorVariances.filter(v => 
      ['Authorization Issues', 'Medical Necessity', 'Coding Errors'].includes(v.rootCause)
    );
    const denialRate = (denialVariances.length / payorVariances.length * 100) || 0;
    
    // Net Collection Rate
    const totalBilled = payorVariances.reduce((sum, v) => sum + v.billedAmount, 0);
    const totalVariance = payorVariances.reduce((sum, v) => sum + v.varianceAmount, 0);
    const totalWriteOff = payorWriteOffs.reduce((sum, w) => sum + w.writeOffAmount, 0);
    const netCollectionRate = ((totalBilled - totalVariance - totalWriteOff) / totalBilled * 100) || 0;
    
    // Days in A/R: Use aging bucket midpoints
    const agingMidpoints = {
      '0-30 days': 15,
      '31-60 days': 45,
      '61-90 days': 75,
      '91-120 days': 105,
      '121-180 days': 150,
      '181-365 days': 273,
      '365+ days': 400
    };
    const openVariances = payorVariances.filter(v => 
      ['Open', 'In Progress', 'Under Appeal'].includes(v.status)
    );
    const totalAgingDays = openVariances.reduce((sum, v) => 
      sum + (agingMidpoints[v.agingBucket] || 0), 0
    );
    const daysInAR = openVariances.length > 0 
      ? Math.round(totalAgingDays / openVariances.length) 
      : 0;
    
    // Clean Claim Rate: Resolved without write-off or extended aging
    const cleanClaims = payorVariances.filter(v => 
      v.status === 'Resolved' && 
      !['181-365 days', '365+ days'].includes(v.agingBucket)
    );
    const cleanClaimRate = (cleanClaims.length / payorVariances.length * 100) || 0;
    
    // Contract Compliance: Inverse of contractual dispute rate
    const contractualVariances = payorVariances.filter(v => 
      v.rootCause === 'Contractual Disputes'
    );
    const contractCompliance = 100 - (contractualVariances.length / payorVariances.length * 100) || 100;
    
    return {
      payorName: payor.payorName,
      payorType: payor.payorType,
      denialRate: denialRate.toFixed(1),
      netCollectionRate: netCollectionRate.toFixed(1),
      daysInAR: daysInAR,
      cleanClaimRate: cleanClaimRate.toFixed(1),
      contractCompliance: contractCompliance.toFixed(1),
      overallScore: calculateOverallScore(denialRate, netCollectionRate, daysInAR, cleanClaimRate, contractCompliance),
      // Additional context from existing data
      varianceCount: payorVariances.length,
      writeOffCount: payorWriteOffs.length,
      totalVarianceAmount: totalVariance,
      totalWriteOffAmount: totalWriteOff,
      openVarianceCount: openVariances.length
    };
  });
};

4. Color Coding Thresholds
javascriptconst getStatusColor = (metric, value) => {
  const thresholds = {
    denialRate: { green: 5, yellow: 10 },        // < 5% green, 5-10% yellow, > 10% red
    netCollectionRate: { green: 96, yellow: 93 }, // > 96% green, 93-96% yellow, < 93% red
    daysInAR: { green: 40, yellow: 60 },          // < 40 green, 40-60 yellow, > 60 red
    cleanClaimRate: { green: 95, yellow: 90 },    // > 95% green, 90-95% yellow, < 90% red
    contractCompliance: { green: 98, yellow: 95 } // > 98% green, 95-98% yellow, < 95% red
  };
  
  const t = thresholds[metric];
  if (metric === 'denialRate' || metric === 'daysInAR') {
    // Lower is better
    if (value < t.green) return 'green';
    if (value < t.yellow) return 'yellow';
    return 'red';
  } else {
    // Higher is better
    if (value > t.green) return 'green';
    if (value > t.yellow) return 'yellow';
    return 'red';
  }
};

const calculateOverallScore = (denial, collection, ar, clean, compliance) => {
  let greenCount = 0;
  let redCount = 0;
  
  if (denial < 5) greenCount++; else if (denial > 10) redCount++;
  if (collection > 96) greenCount++; else if (collection < 93) redCount++;
  if (ar < 40) greenCount++; else if (ar > 60) redCount++;
  if (clean > 95) greenCount++; else if (clean < 90) redCount++;
  if (compliance > 98) greenCount++; else if (compliance < 95) redCount++;
  
  if (greenCount === 5) return 'A';
  if (greenCount === 4) return 'B+';
  if (greenCount === 3) return 'B';
  if (greenCount === 2 && redCount === 0) return 'B-';
  if (greenCount === 2) return 'C+';
  if (greenCount === 1 && redCount <= 1) return 'C';
  if (greenCount === 1) return 'C-';
  if (redCount <= 2) return 'D';
  return 'F';
};

javascriptconst generateAlerts = (variances, writeOffs, payors) => {
  const alerts = [];
  
  // Check each payor's metrics against thresholds
  payors.forEach(payor => {
    const metrics = calculatePayorMetrics(
      variances.filter(v => v.payorName === payor.payorName),
      writeOffs.filter(w => w.payorName === payor.payorName)
    );
    
    // High denial rate alert
    if (metrics.denialRate > 12) {
      const denialCount = variances.filter(v => 
        v.payorName === payor.payorName &&
        ['Authorization Issues', 'Medical Necessity'].includes(v.rootCause) &&
        v.status === 'Open'
      ).length;
      
      alerts.push({
        type: 'warning',
        payor: payor.payorName,
        message: `Denial rate at ${metrics.denialRate}% - ${denialCount} open denial variances`,
        action: 'Review denial patterns'
      });
    }
    
    // High A/R alert
    const oldARCount = variances.filter(v => 
      v.payorName === payor.payorName &&
      ['121-180 days', '181-365 days', '365+ days'].includes(v.agingBucket) &&
      v.status !== 'Resolved' && v.status !== 'Written Off'
    ).length;
    
    if (oldARCount > 20) {
      alerts.push({
        type: 'critical',
        payor: payor.payorName,
        message: `${oldARCount} variances over 120 days old`,
        action: 'Prioritize aged receivables'
      });
    }
    
    // Underpayment opportunity alert
    const underpaymentAmount = variances
      .filter(v => 
        v.payorName === payor.payorName &&
        v.rootCause === 'Contractual Disputes' &&
        v.status !== 'Written Off'
      )
      .reduce((sum, v) => sum + v.varianceAmount, 0);
    
    if (underpaymentAmount > 50000) {
      alerts.push({
        type: 'opportunity',
        payor: payor.payorName,
        message: `$${underpaymentAmount.toLocaleString()} in underpayment recovery opportunity`,
        action: 'Initiate contract review'
      });
    }
  });
  
  return alerts.sort((a, b) => {
    const priority = { critical: 0, warning: 1, opportunity: 2 };
    return priority[a.type] - priority[b.type];
  });
};
