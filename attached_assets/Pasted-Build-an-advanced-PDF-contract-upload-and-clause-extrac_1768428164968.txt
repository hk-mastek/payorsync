3. def find_text_positions(
     search_text: str,
     page_texts: list,
     expected_page: int = None
   ) -> List[dict]:
   """
   Find bounding box coordinates for text in PDF
   For highlighting in the viewer
   """
   
   bounding_boxes = []
   
   # Clean and normalize search text
   search_text_clean = ' '.join(search_text.split())
   
   # Search in expected page first, then others
   pages_to_search = [expected_page] if expected_page else range(1, len(page_texts) + 1)
   
   for page_info in page_texts:
       if page_info['page_number'] not in pages_to_search:
           continue
       
       # Search through blocks
       for block in page_info.get('blocks', []):
           if isinstance(block, (list, tuple)) and len(block) >= 5:
               x0, y0, x1, y1, block_text = block[:5]
               block_text_clean = ' '.join(block_text.split())
               
               # Check if search text is in this block
               if search_text_clean[:100] in block_text_clean or \
                  block_text_clean in search_text_clean:
                   bounding_boxes.append({
                       'page': page_info['page_number'],
                       'x1': float(x0),
                       'y1': float(y0),
                       'x2': float(x1),
                       'y2': float(y1)
                   })
   
   return bounding_boxes

4. HELPER FUNCTIONS:

def calculate_avg_confidence(ocr_data: dict) -> float:
    """Calculate average OCR confidence score"""
    confidences = [float(conf) for conf in ocr_data['conf'] if conf != '-1']
    return sum(confidences) / len(confidences) if confidences else 0.0

def extract_json_from_response(response: str) -> str:
    """Extract JSON from Claude response, handling markdown code blocks"""
    # Remove ```json and ``` if present
    json_str = response.strip()
    if json_str.startswith('```json'):
        json_str = json_str[7:]
    if json_str.startswith('```'):
        json_str = json_str[3:]
    if json_str.endswith('```'):
        json_str = json_str[:-3]
    return json_str.strip()

def split_text_into_chunks(text: str, chunk_size: int) -> List[str]:
    """Split long text into processable chunks"""
    # Smart splitting on section boundaries
    sections = re.split(r'\n\s*(?:SECTION|Section|ARTICLE|Article)\s+\d+', text)
    
    chunks = []
    current_chunk = ""
    
    for section in sections:
        if len(current_chunk) + len(section) < chunk_size:
            current_chunk += section
        else:
            if current_chunk:
                chunks.append(current_chunk)
            current_chunk = section
    
    if current_chunk:
        chunks.append(current_chunk)
    
    return chunks

    uploaded_contract = await create_uploaded_contract(
        original_filename=file.filename,
        file_path=file_path,
        file_size=file.size,
        mime_type=file.content_type,
        uploaded_by=current_user.id,
        payor_id=payor_id,
        contract_id=contract_id,
        processing_status='pending'
    )
    
    # Queue background processing
    background_tasks.add_task(
        process_uploaded_pdf,
        file_path=file_path,
        uploaded_contract_id=uploaded_contract.id,
        user_id=current_user.id
    )
    
    return {
        "upload_id": uploaded_contract.id,
        "status": "processing",
        "message": "PDF upload successful. Processing has started."
    }


@router.get("/upload/{upload_id}/status")
async def get_processing_status(
    upload_id: UUID,
    current_user: User = Depends(get_current_user)
):
    """
    Check processing status of uploaded PDF
    """
    
    uploaded_contract = await get_uploaded_contract(upload_id)
    
    if not uploaded_contract:
        raise HTTPException(status_code=404, detail="Upload not found")
    
    # Check authorization
    if uploaded_contract.uploaded_by != current_user.id and not current_user.is_admin:
        raise HTTPException(status_code=403, detail="Access denied")
    
    response = {
        "upload_id": uploaded_contract.id,
        "filename": uploaded_contract.original_filename,
        "status": uploaded_contract.processing_status,
        "total_pages": uploaded_contract.total_pages,
        "total_clauses_extracted": uploaded_contract.total_clauses_extracted,
        "uploaded_at": uploaded_contract.uploaded_at,
        "completed_at": uploaded_contract.completed_at
    }
    
    if uploaded_contract.processing_status == 'failed':
        response['error'] = uploaded_contract.error_message
    
    if uploaded_contract.processing_status == 'completed':
        response['metadata'] = uploaded_contract.metadata
        response['clauses_url'] = f"/api/pdf/upload/{upload_id}/clauses"
    
    return response
